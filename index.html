import requests

cve_id = "CVE-2026-63030"
url = f"https://api.vinthub.io/cve/{cve_id.split('-')[1]}/{cve_id}.json"

response = requests.get(url)
data = response.json()

print(f"SSVC Decision: {data['ssvc']['decision']}")
print(f"EPSS Probability: {data['epss']['score']}")
print(f"PoCs Found: {len(data['pocs'])}")
